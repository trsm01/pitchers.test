// 【改訂版】個人記録の推移グラフを描画・更新する処理 function updateTrendChart(selectedName, selectedEvent) { // 1. 選択された選手のみのデータを抽出（名前の前後スペースも無視して完全一致させる） const targetName = selectedName.trim(); let playerRecords = csvData.filter(row => row['氏名'] && row['氏名'].trim() === targetName); // 2. 測定月の空欄や不正データを弾き、安全なデータだけを残す playerRecords = playerRecords.filter(row => row['測定月'] && row[selectedEvent]); // 3. 【重要】「測定月」を文字ではなく「日付（Dateオブジェクト）」として時系列順に正しく並べ替える playerRecords.sort((a, b) => { // "2026-9" などを "2026/09/01" の形に変換して比較（2026-10バグを防止） const dateA = new Date(a['測定月'].replace(/-/g, '/') + '/01'); const dateB = new Date(b['測定月'].replace(/-/g, '/') + '/01'); return dateA - dateB; }); // 4. グラフ用のラベル（X軸：測定月）とデータ（Y軸：記録）の配列を作成 const labels = playerRecords.map(row => row['測定月'].trim()); const dataPoints = playerRecords.map(row => Number(row[selectedEvent])); // 5. Chart.jsでグラフを描画・更新する（変数名 chartInstance や trendChart はご自身の環境に合わせてください） if (window.trendChart) { window.trendChart.destroy(); // 既存のグラフを一旦リセット } const ctx = document.getElementById('trendChartCanvas').getContext('2d'); window.trendChart = new Chart(ctx, { type: 'line', // 推移なので折れ線グラフ data: { labels: labels, datasets: [{ label: `${targetName}選手の${selectedEvent}推移`, data: dataPoints, borderColor: 'rgba(54, 162, 235, 1)', backgroundColor: 'rgba(54, 162, 235, 0.2)', borderWidth: 2, tension: 0.1 // 直線を少し滑らかにする }] }, options: { responsive: true, scales: { y: { beginAtZero: false // 記録の変動を見やすくするため0からスタートさせない } } } }); }
